% \exfile{relative-path}{language} — verbatim, compiled-and-run source.
\newcommand{\exfile}[2]{%
  \lstinputlisting[language=#2]{#1}}

% \resultfile{results/file.(json|csv)}{caption} — the actual generated
% result, shown verbatim (monospace, framed). If it has not been
% regenerated yet, a placeholder is shown so the PDF still compiles.
\newcommand{\resultfile}[2]{%
  \par\medskip\noindent\textbf{Result.} \textit{#2}\par\nopagebreak
  \IfFileExists{#1}{%
    \lstinputlisting[basicstyle=\ttfamily\scriptsize,numbers=none,
      backgroundcolor=\color{bg},frame=single]{#1}}{%
    \begin{center}\fbox{\small\itshape
      not yet generated — run \texttt{docs/latex/regen.sh}}\end{center}}%
  \medskip}
